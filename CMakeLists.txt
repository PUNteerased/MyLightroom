cmake_minimum_required(VERSION 3.24)
project(MyLightroom VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

option(MYLR_USE_LIBRAW "Enable LibRaw CR2/CR3 decoding" ON)
option(MYLR_USE_ONNX "Enable ONNX Runtime AI inference" ON)
option(MYLR_BUILD_TESTS "Build unit tests" ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets Sql Concurrent)

if(MYLR_BUILD_TESTS)
    find_package(Qt6 REQUIRED COMPONENTS Test)
endif()

if(MYLR_USE_LIBRAW)
    find_package(libraw CONFIG QUIET)
    if(libraw_FOUND)
        add_compile_definitions(MYLR_HAS_LIBRAW=1)
        set(LibRaw_FOUND TRUE)
    else()
        find_package(LibRaw QUIET)
        if(LibRaw_FOUND)
            add_compile_definitions(MYLR_HAS_LIBRAW=1)
        else()
            message(WARNING "LibRaw not found — RAW decode uses preview fallback")
            add_compile_definitions(MYLR_HAS_LIBRAW=0)
            set(MYLR_USE_LIBRAW OFF)
        endif()
    endif()
else()
    add_compile_definitions(MYLR_HAS_LIBRAW=0)
endif()

if(MYLR_USE_ONNX)
    find_package(onnxruntime QUIET)
    if(onnxruntime_FOUND)
        add_compile_definitions(MYLR_HAS_ONNX=1)
    else()
        message(WARNING "ONNX Runtime not found — AI Stage 2 uses heuristic fallback")
        add_compile_definitions(MYLR_HAS_ONNX=0)
        set(MYLR_USE_ONNX OFF)
    endif()
else()
    add_compile_definitions(MYLR_HAS_ONNX=0)
endif()

add_subdirectory(third_party)
add_subdirectory(src)

if(MYLR_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

install(TARGETS MyLightroom RUNTIME DESTINATION bin)
install(DIRECTORY presets/ DESTINATION share/MyLightroom/presets)
install(DIRECTORY profiles/ DESTINATION share/MyLightroom/profiles)
install(DIRECTORY models/ DESTINATION share/MyLightroom/models)
